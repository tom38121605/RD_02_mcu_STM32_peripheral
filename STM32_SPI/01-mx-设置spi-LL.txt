      uint8_t sconfigbuf[30]={0};
      uint8_t iadcdata[ADCDATALEN]={0};

      const uint8_t mcp3561_config[25]=      
      {       
          0x82,  //CONFIG0 , internal 2.4v, external clk, standby, null
          0x0C,  //CONFIG1 ,  prescaler 1x, OSR 256 
          0xA3,  //CONFIG2,    //gain 8x，BOOST 1x，enable chop
          0x83,  //CONFIG3, One-shot, (24bit)
          0x72,  //IRQ, pin output IRQ  , Hi-Z no startadc irq     
          0x01,              //MUX , CH0(v+),CH1 (v-)          
          0xE0,0x00,0x00,    //SCAN  512 dmclk, scan none    
          0x00,0x00,0x00,    //TIMER       
          0x00,0x00,0x00,    //OFFSETCAL        
          0x80,0x00,0x00,     //GAINCAL         
          0x90,0x00,0x00,     //RESERVED    
          0x30,                 //RESERVED    
          0xA5,                 //LOCK
          0x00,0x0C,           //RESERVED    
      };  

      //---------------------mcp3561 module---------------------------
       
      void MCP3561_CS_LOW(void)
      {
         LL_GPIO_ResetOutputPin(SPI1_CS_GPIO_Port, SPI1_CS_Pin);
      }

      void MCP3561_CS_HIGH(void)
      {
         LL_GPIO_SetOutputPin(SPI1_CS_GPIO_Port, SPI1_CS_Pin);
      }
       
      void mcp3561_write(uint8_t* pbuf, uint8_t icmd, uint8_t inum)
      {         
         MCP3561_CS_LOW();         
         SPI_transfer(icmd);
         while (inum--)
         {
            SPI_transfer(*pbuf);
            pbuf++;
         }
         MCP3561_CS_HIGH(); 
      }

      void mcp3561_read(uint8_t* pbuf, uint8_t icmd, uint8_t inum)
      {	
         MCP3561_CS_LOW();
         SPI_transfer(icmd);
         while (inum--)
         {
            *pbuf = SPI_transfer(DUMMY_BYTE);   //0xFF
            pbuf++;
         }         
         MCP3561_CS_HIGH(); 
      }

      void mcp3561_init(void)
      { 
         //set mcp3561 config
         mcp3561_write( (uint8_t*)mcp3561_config ,CMD_WRITE_CONFIG0,12);            
       } 

      void mcp3561_adcstart(void)
      {       
         MCP3561_CS_LOW();
         SPI_transfer(FASTCMD_ADCSTART);    //start adc           
         MCP3561_CS_HIGH();        
      } 

      void mcp3561_adcshutdown(void)
      {       
         MCP3561_CS_LOW();
         SPI_transfer(FASTCMD_ADCSHUTDOWN);    //shutdown adc           
         MCP3561_CS_HIGH();        
      } 

      uint8_t mcp3561_getstatus(void)
      {
         uint8_t istatus=0;         
         MCP3561_CS_LOW();
         istatus=SPI_transfer(FASTCMD_UNUSE);   // UNUSE cmd            
         MCP3561_CS_HIGH(); 
         return istatus;
      }


      //---------------------mcp3561 module----end----------------------
