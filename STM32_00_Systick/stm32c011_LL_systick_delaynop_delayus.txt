=======================systick 延时1ms===推导过程=======================

Systick初始化过程：
        LOAD  =  ((HCLKFrequency / Ticks) - 1 );     //如 1ms:  48M / 1000  //如 1ms:  48M / 1000 -1
        CTRL.2,0=1,1 ，时钟源选AHB（48M），使能systick

Systick中断触发过程：
     CTRL.2=1 使能systick  --  （LOAD 的值赋给 -- VAL）  --  VAL不断减1（每个clk都减1）  --   直到VAL=0 -- 出现中断标志 CRTL.16=1 

main()

    LL_Init1msTick(48000000);
         LL_InitTick(48M, 1000);


--------------------
 
功能：设置systick为1ms延时
参数:  48M, 1000
 LL_InitTick( HCLKFrequency,  Ticks)
 
     SysTick->LOAD  =  ((HCLKFrequency / Ticks) - 1 );                                       //如 1ms:  48M / 1000 -1
     SysTick->VAL   = 0 ;                                                                                 //清零计数器                                                          
     SysTick->CTRL  = SysTick_CTRL_CLKSOURCE_Msk |  *_CTRL_ENABLE_Msk;    //CTRL.2,0=1,1 ，时钟源选AHB（48M），使能systick

=======================systick 延时1us===函数=======================


//时钟频率为48M
//       delay_us(1);        // 1.6us 
//       delay_us(2);        // 2.6us
//       delay_us(5);        // 5.7us 
//       delay_us(10);       // 10.8us
//       delay_us(20);       // 20.8us
//       delay_us(30);       // 31.2us
//       delay_us(50);       // 51.6us
//       delay_us(100);      // 104.4us 
void delay_us(uint32_t us)
{
    uint32_t idelaytick;
    uint32_t iold, inow, itickcount = 0;

    uint32_t reload = SysTick->LOAD;   // 获取SysTick重载值（系统时钟/1000000）
    
    idelaytick = us * (48000000 / 1000000);    // 计算需要的SysTick时钟节拍数
 
    iold = SysTick->VAL;      // 记录当前SysTick计数值
    
    while (1)
    {
        inow = SysTick->VAL;
        if (inow != iold)
        {
 
            // SysTick是递减计数器，计算已走过的节拍
            if (inow < iold) 
               itickcount += iold - inow;
            else 
               itickcount += reload - inow + iold;               
            
            iold = inow;
 
            if (itickcount >= idelaytick) break;   // 达到目标节拍数则退出
        }
    }
}

//       delay_nop(100);     // 2.8us
//       delay_nop(90);      // 2.6us         
//       delay_nop(80);      // 2.3us         
//       delay_nop(70);      // 2.0us
//       delay_nop(35);      // 1.1us
void delay_nop(uint32_t nCount) 
{ 
   for(; nCount != 0; nCount--)  
   { 
      __NOP(); 
   } 
}

//注意 nms 的范围
//nms<=0xffff*8*1000/SYSCLK
void LL_mDelay(uint32_t Delay)
{
   uint32_t temp;   
   SysTick->LOAD=(uint32_t)Delay*(SystemCoreClock/1000/8); 
   SysTick->VAL =0x00;            
   SysTick->CTRL=0x01 ;          
   
   do
   {
      temp=SysTick->CTRL;
   }
   while((temp&0x01)&&!(temp&(1<<16))); 
   
   SysTick->CTRL=0x00;         
   SysTick->VAL =0X00;           
}

void LL_uDelay(uint32_t Delay)
{
   uint32_t temp;   
   SysTick->LOAD=(uint32_t)Delay*(SystemCoreClock/1000000/8); 
   SysTick->VAL =0x00;            
   SysTick->CTRL=0x01 ;          
   
   do
   {
      temp=SysTick->CTRL;
   }
   while((temp&0x01)&&!(temp&(1<<16))); 
   
   SysTick->CTRL=0x00;         
   SysTick->VAL =0X00;           
}
