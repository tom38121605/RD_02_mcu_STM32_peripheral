
-------定时器时间计算----250M----------

   1).  定时 1ms 
          。Prescaler：  249                          //1Mhz  = 250M / 250
          。Couter Period：  999                  //1khz    =  1M / 1000  

   2).  定时 10ms 
         。Prescaler：  249                          //1Mhz  = 250M / 250
         。Couter Period：  9999                 //100hz    =  1M / 10000  

   3).  定时 1us  
         。Prescaler：  24                       //10Mhz  = 250M / 25
         。Couter Period：  9                 //1Mhz    =  10M / 10 


-------定时器时间计算----48M----------

   1).  定时 1ms 
          。Prescaler：  47                           //1Mhz  = 48M / 48
          。Couter Period：  999                  //1khz    =  1M / 1000  



MCU: H503   (其他同系列mcu类似)

一. 用mx添加timer3步骤：  （？20个地方）

1. 建立工程     （4个地方）
   。双击打开mx
   。选择 start my project from mcu
   。输入mcu型号： STM32F030CCT6
   。点击start project

2. 设置时钟为HSI的250M     （1个地方）

3. 设置TIM3的频率为1KHz， 如下：    （8个地方）

    。勾选： Internal Clock

    。Prescaler：  249                          //1Mhz  = 250M / 250
    。Couter Period：  999                   //1khz   =  1M / 1000  
    。Auto-reload preload: disable

    。timer3 interrupt：  勾选     

5. 设置project，如下：  （6个地方）

    。设置工程目录： d:\test_timer3
    。设置工程名称： prjtim3
    。设置IDE: MDK-ARM

    。勾选 copy only the necessary library files
    。勾选 Generate peripheral initialiation as a pair of 'c/h"files

    。选择 LL库


6. 保存mx，点击generate code，生成kel工程， ok      //自动生成如下（文件和代码添加） 

    1) 自动添加如下文件
         Src/tim.c
         Inc/tim.h
         Drivers/STM32H5xx_HAL_Driver/Src/stm32h5xx_ll_tim.c
         Drivers/STM32H5xx_HAL_Driver/Inc/stm32h5xx_ll_tim.h



二.  在keil 工程中，添加代码 


--- tim.h  ----

   void MX_TIM3_Init(void);   // MX自动添加


--- tim.c  ----

   void MX_TIM3_Init(void)      // MX自动添加

      ......


     /* USER CODE BEGIN TIM3_Init 2 */

      LL_TIM_ClearFlag_UPDATE(TIM3);     //手动添加
      LL_TIM_EnableIT_UPDATE(TIM3);      //手动添加
      LL_TIM_EnableCounter(TIM3);          //手动添加

     /* USER CODE END TIM3_Init 2 */


----- stm32h5xx_it.h ----- 

   void TIM3_IRQHandler(void);   // MX自动添加 


----- stm32h5xx_it.c -----

   void TIM3_IRQHandler(void)     // MX自动添加 
   {
       /* USER CODE BEGIN TIM3_IRQn 0 */
  

        if(LL_TIM_IsActiveFlag_UPDATE(TIM3) == 1)       //手动添加
       {
           LL_TIM_ClearFlag_UPDATE(TIM3);
       }

       TimerUpdate_Callback();         //手动添加
       //flg_timer3 = 1;               //手动添加，如果不想用callback，直接用这里的flg即可

       /* USER CODE END TIM3_IRQn 0 */
       /* USER CODE BEGIN TIM3_IRQn 1 */

       /* USER CODE END TIM3_IRQn 1 */
   }



--- main.h ----

   #include "stm32h5xx_ll_tim.h"        // MX自动添加 


   extern volatile uint8_t flg_timer3;      //手动添加
   void TimerUpdate_Callback(void);         //手动添加



--- main.c  ----

   #include "tim.h"                        // MX自动添加 

   volatile uint8_t flg_timer3 = 0;    //手动添加

   void TimerUpdate_Callback(void)     //手动添加   //考虑在mx的project中自动添加 （LL库不能自动添加）
  {
       flg_timer3 = 1;
   }


   main()

     MX_TIM3_Init();         // MX自动添加 


    while (1)    //手动添加
    
  
       static uint32_t itimercount=0;
  
      
       if(flg_timer3==1)
       {
          flg_timer3=0;
          itimercount++;
       }
             
       if(itimercount>=1)
       {
          itimercount=0;
          //printf( "r3\r\n");
          LED_Toggle();
      }
