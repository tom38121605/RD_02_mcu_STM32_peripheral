一. 在 LL库中添加 uart rx dma中断步骤：     (在uart tx dma的基础上添加)

1. 已建立uart简单的 tx工程     （非dma）                     
   。可printf发送字符串
   。可iputs发送字符数组
   。发送完成后，触发uart的发送完成TC中断，小灯点亮  //可选
 
 
2. 添加 uart rx dma  （ch0）
    。在mx的选项卡System core中， 点选GPDMA1 
    。在上面，选中channel1 ，从下拉框中选择“Standard Request"

    。在下面，选中CH0:
          . Circular Mode :       Enable                  
          . Request          :       USART1_RX

          .  Priority          :       High
          .  Direction       :       Peripheral  to Mem
          . Source Address Inc...        :    Disable     （默认）
          . Destination Address Inc...  :    Enable

          . 其它都是默认

3. 设置project，如下：  （6个地方）

    。设置工程目录： 原有的不变
    。设置工程名称： 原有的不变
    。设置IDE: MDK-ARM

    。勾选 copy only the necessary library files                            //原有的不变
    。勾选 Generate peripheral initialiation as a pair of 'c/h"files   //原有的不变

    。新加的DMA选择LL库


4. 保存mx，点击generate code，生成kel工程， ok

      //--- 设置dma数据，开启dma，使能uart的dma的发送 ---

       //传输数据长度
       MODIFY_REG(GPDMA1_Channel0->CBR1, DMA_CBR1_BNDT, (TXBUFFERSIZE & DMA_CBR1_BNDT));
           //GPDMA1_Channel0->CBR1 = 6;  

       //清0所有CFCR中的下列中断标志
       SET_BIT(GPDMA1_Channel0->CFCR, DMA_CSR_TCF | DMA_CSR_HTF | DMA_CSR_DTEF | DMA_CSR_ULEF | DMA_CSR_USEF | DMA_CSR_SUSPF | DMA_CSR_TOF);
           //GPDMA1_Channel0->CFCR =  1<<8 |1<<9 | 1<<10 | 1<<11 | 1<<12 | 1<<13 | 1<<14; 

       //设置通道1的源数据地址 
       WRITE_REG(GPDMA1_Channel0->CSAR, (uint32_t)aTxBuffer);
         //GPDMA1_Channel0->CSAR = (uint32_t)aTxBuffer;    

       //设置通道1的目标数据地址
       WRITE_REG(GPDMA1_Channel0->CDAR, (uint32_t)&USART1->TDR);
         //GPDMA1_Channel0->CDAR = (uint32_t)&USART1->TDR;    
          
        //使能完成中断,传输err中断，更新链接err中断，用户设置err中断，溢出中断
       SET_BIT(GPDMA1_Channel0->CCR, DMA_CCR_TCIE | DMA_CCR_DTEIE | DMA_CCR_ULEIE | DMA_CCR_USEIE | DMA_CCR_TOIE);
          //GPDMA1_Channel0->CCR = 1<<8 | 1<<10 | 1<<11 | 1<<12 | 1<<14;                        

       //使能半完成中断
       SET_BIT(GPDMA1_Channel0->CCR, DMA_CCR_HTIE);
          //GPDMA1_Channel0->CCR  |=1<<9;                              

       //开启 dma ch0
       LL_DMA_EnableChannel(GPDMA1, LL_DMA_CHANNEL_0);
          //GPDMA1_Channel0->CCR  |=1<<0;                  

       //清0传输完成的中断标志
       SET_BIT(USART1->ICR, USART_ICR_TCCF);
         //USART1->ICR |= 1<<6; 

       //使能uart的dma的发送  
       SET_BIT(USART1->CR3, USART_CR3_DMAT);
          //USART1->CR3=1<<7;     
  

//2. 在stm32h5xx_it.c中加入

      void GPDMA1_Channel0_IRQHandler(void)
      {        
         //HT 
        if(READ_BIT(GPDMA1_Channel0->CSR, 1<<9))
        {
             GPDMA1_Channel0->CFCR = 1<<9;
        }        
        
         //TC 
        if(READ_BIT(GPDMA1_Channel0->CSR, 1<<8))
        {        
           GPDMA1_Channel0->CFCR = 1<<8 | 1<<9;
        }

     }
